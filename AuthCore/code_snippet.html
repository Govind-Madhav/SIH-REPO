server.js - Auth Module Initialization
import express from 'express';
import { Auth } from './auth-engine/index.js';
import { SqliteAdapter } from './auth-engine/adapters/sqlite.js';

const app = express();
app.use(express.json());

// Initialize the Universal Auth Engine
const authSystem = Auth.init({
    storageAdapter: new SqliteAdapter('auth.db'),
    claimsResolver: async ({ userId, context }) => ({
        roles: ['user', 'admin'], 
        tenant: context.tenant 
    }),
    jwtSecret: process.env.JWT_SECRET,
    accessExpiry: '15m',
    refreshExpiryMs: 7 * 24 * 60 * 60 * 1000
});

// Mount the decoupled auth routes
app.use('/auth', authSystem.router);

// Protect your product routes easily
app.get('/api/protected', authSystem.authenticate, (req, res) => {
    res.json({ message: 'Success!', user: req.identity });
});

app.listen(3000, () => console.log('API is live!'));
